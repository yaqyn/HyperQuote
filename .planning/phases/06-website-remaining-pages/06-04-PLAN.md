**chat.ts (SSE streaming server function):**
Create `apps/website/src/lib/chat.ts`. This server function returns a **real SSE streaming Response using ReadableStream**, even though the AI responses are mocked. This validates the SSE transport layer so Phase 30 only needs to swap the AI model call.

```typescript
import { createServerFn } from '@tanstack/react-start'
import { z } from 'zod'

const chatInput = z.object({
  messages: z.array(z.object({
    role: z.enum(['user', 'assistant']),
    content: z.string(),
  })),
})

// Mock responses for development
const MOCK_RESPONSES: Record<string, string> = {
  default: "I'm HyperQuote's assistant. I can help you find building materials, get quotes, and track orders. What are you looking for today?",
  quote: "I'd be happy to help you get a quote! You can start by telling me what materials you need, or use our Material List Builder for a more detailed request.",
  price: "We don't publish exact prices -- they depend on quantity, delivery location, and current market conditions. Request a quote and you'll get a response within 4 hours!",
  delivery: "Delivery times depend on your location and the materials ordered. Most orders within Cairo are delivered within 24-48 hours. Would you like to place an order?",
  cement: "We carry a full range of cement products including Portland CEM I 42.5N, CEM II, and specialty cements. Would you like to see our catalog or get a quote?",
}

function getMockResponse(userMessage: string): string {
  const lower = userMessage.toLowerCase()
  if (lower.includes('quote') || lower.includes('عرض سعر')) return MOCK_RESPONSES.quote
  if (lower.includes('price') || lower.includes('سعر')) return MOCK_RESPONSES.price
  if (lower.includes('deliver') || lower.includes('توصيل')) return MOCK_RESPONSES.delivery
  if (lower.includes('cement') || lower.includes('اسمنت')) return MOCK_RESPONSES.cement
  return MOCK_RESPONSES.default
}

// Return a streaming SSE Response using ReadableStream
// The mock simulates token-by-token streaming with delays
export const chatStream = createServerFn()
  .inputValidator(chatInput)
  .handler(async ({ data: input }) => {
    const lastMessage = input.messages[input.messages.length - 1]
    const response = getMockResponse(lastMessage?.content ?? '')
    const words = response.split(' ')

    // Create a ReadableStream that emits SSE events
    const stream = new ReadableStream({
      async start(controller) {
        const encoder = new TextEncoder()
        for (const word of words) {
          // SSE format: data: {"content":"word "}\n\n
          const chunk = JSON.stringify({ content: word + ' ' })
          controller.enqueue(encoder.encode(`data: ${chunk}\n\n`))
          // Simulate token delay (50-100ms)
          await new Promise(r => setTimeout(r, 50 + Math.random() * 50))
        }
        controller.enqueue(encoder.encode('data: [DONE]\n\n'))
        controller.close()
      }
    })

    return new Response(stream, {
      headers: {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache',
        'Connection': 'keep-alive',
      },
    })
  })
```

**useAIChat.ts (abstraction layer using real fetchServerSentEvents):**
This file is the ONLY file that imports from `@tanstack/ai-react`. It wraps `useChat` + `fetchServerSentEvents` per RESEARCH.md Pattern 3.

```typescript
import { fetchServerSentEvents, useChat } from '@tanstack/ai-react'

interface ChatOptions {
  onError?: (error: Error) => void
}

export function useAIChat(options?: ChatOptions) {
  const chat = useChat({
    // fetchServerSentEvents handles SSE parsing, reconnection, [DONE] marker
    connection: fetchServerSentEvents('/api/chat'),
    onError: options?.onError,
  })

  return {
    messages: chat.messages,
    sendMessage: chat.sendMessage,
    isLoading: chat.isLoading,
    error: chat.error,
    clear: chat.clearMessages ?? (() => {}),
  }
}
```

The URL `/api/chat` should match wherever the SSE endpoint is exposed. If using the `chatStream` server function directly, adjust the connection URL accordingly. The abstraction ensures consumers (ChatPanel, ChatMessages) never import from `@tanstack/ai-react` directly.

interface ChatWidgetState {
  isOpen: boolean
  hasSeenPulse: boolean
  open: () => void

  dismissPulse: () => void

export const useChatWidget = create<ChatWidgetState>((set) => ({

  hasSeenPulse: false,
  open: () => set({ isOpen: true, hasSeenPulse: true }),
