    Structure:
    ```tsx
    import { createFileRoute } from '@tanstack/react-router'
    import { createServerFn } from '@tanstack/react-start'
    import { useState } from 'react'
    import { useTranslation } from 'react-i18next'
    import {
      GlassWindow, CurrencyDisplay, DateDisplay, UnitDisplay,
      StatusBadge, Skeleton, EmptyState, LionMark, cn
    } from '@hyperquote/ui'

    const getMockProducts = createServerFn({ method: 'GET' }).handler(async () => {
      return [
        { id: '1', name: 'Portland Cement', nameAr: 'أسمنت بورتلاندي', price: 56400, unit: 'ton', qty: 500 },
        { id: '2', name: 'Reinforcing Steel', nameAr: 'حديد تسليح', price: 32000, unit: 'ton', qty: 150 },
        { id: '3', name: 'Sand', nameAr: 'رمل', price: 850, unit: 'cubic_meter', qty: 2500 },
      ]
    })

    export const Route = createFileRoute('/vertical-slice')({
      loader: () => getMockProducts(),
      component: VerticalSlicePage,
    })

    function VerticalSlicePage() {
      const products = Route.useLoaderData()
      const { t, i18n } = useTranslation()
      const [glassOpen, setGlassOpen] = useState(false)
      const isAr = i18n.language === 'ar'

      const switchLang = (lang: 'ar' | 'en') => {
        i18n.changeLanguage(lang)
        document.documentElement.lang = lang
        document.documentElement.dir = lang === 'ar' ? 'rtl' : 'ltr'
      }

      const toggleTheme = () => {
        const html = document.documentElement
        const current = html.getAttribute('data-theme')
        html.setAttribute('data-theme', current === 'dark' ? 'light' : 'dark')
      }

      return (
        <div className="relative min-h-screen p-8">
          <LionMark className="fixed inset-0 z-0" />
          <div className="relative z-10 flex gap-4 mb-8">
            <button onClick={() => switchLang('ar')} className="bg-[var(--color-primary)] text-white rounded-lg px-4 py-2 font-medium">العربية</button>
            <button onClick={() => switchLang('en')} className="bg-[var(--color-primary)] text-white rounded-lg px-4 py-2 font-medium">English</button>
            <button onClick={toggleTheme} className="bg-[var(--color-surface)] text-[var(--color-text)] rounded-lg px-4 py-2 font-medium">{t('theme')}</button>
            <button onClick={() => setGlassOpen(true)} className="bg-[var(--color-primary)] text-white rounded-lg px-4 py-2 font-medium">Open Glass</button>
          </div>

          <section className="relative z-10 space-y-6">
            <h2 className="text-[var(--text-2xl)] font-semibold">{t('app_name')}</h2>
            <div className="ps-4 border-s-2 border-[var(--color-primary)]">
              <CurrencyDisplay value={56400} />
            </div>
            <div><DateDisplay date={new Date()} /></div>
            {products.map(p => (
              <div key={p.id} className="flex items-center gap-4 py-2">
                <span className="font-medium">{isAr ? p.nameAr : p.name}</span>
                <UnitDisplay value={p.qty} unit={p.unit} />
                <CurrencyDisplay value={p.price} />
                <StatusBadge status="success">Active</StatusBadge>
              </div>
            ))}
            <div className="flex gap-2">
              <StatusBadge status="success">{t('save')}</StatusBadge>
              <StatusBadge status="warning">Pending</StatusBadge>
              <StatusBadge status="error">{t('error')}</StatusBadge>
              <StatusBadge status="info">Info</StatusBadge>
              <StatusBadge status="neutral">Draft</StatusBadge>
            </div>
            <div className="space-y-2">
              <Skeleton className="h-4 w-48" />
              <Skeleton className="h-4 w-32" />
            </div>
            <EmptyState title={t('no_results')} description="Try adjusting filters" />
          </section>

          <GlassWindow isOpen={glassOpen} onClose={() => setGlassOpen(false)}>
            <div className="p-8">
              <h3 className="text-[var(--text-xl)] font-semibold mb-4">Glass Window</h3>
              <p>{t('welcome')}</p>
              <CurrencyDisplay value={123456} />
            </div>
          </GlassWindow>
        </div>
      )
    }
    ```
