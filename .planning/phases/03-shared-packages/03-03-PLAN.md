       ```typescript
       import { redirect } from '@tanstack/react-router'
       import { createSupabaseServerClient } from './server'
       import { getRequest } from '@tanstack/react-start/server'
       import type { AuthGuardOptions, AuthSession } from './types'

       export async function authGuard(opts: AuthGuardOptions): Promise<AuthSession> {
         const request = getRequest()
         const { client } = createSupabaseServerClient({
           request,
           supabaseUrl: opts.supabaseUrl,
           supabaseAnonKey: opts.supabaseAnonKey,
         })
         const { data: { session } } = await client.auth.getSession()

         if (!session) {
           throw redirect({ to: opts.loginPath ?? '/login' })
         }

         return { session, user: session.user }
       }
       ```

       ```typescript
       import { type ReactNode } from 'react'
       import { FormProvider, useForm, type UseFormProps, type FieldValues } from 'react-hook-form'
       import { Form as AriaForm } from 'react-aria-components'

       interface FormRootProps<T extends FieldValues> extends UseFormProps<T> {
         children: ReactNode
         onSubmit: (data: T) => void | Promise<void>
         className?: string
       }

       export function FormRoot<T extends FieldValues>({
         children,
         onSubmit,
         className,
         ...formProps
       }: FormRootProps<T>) {
         const methods = useForm<T>(formProps)
         return (
           <FormProvider {...methods}>
             <AriaForm onSubmit={methods.handleSubmit(onSubmit)} className={className}>
               {children}
             </AriaForm>
           </FormProvider>
         )
       }
       ```
