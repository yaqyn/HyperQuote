       ```typescript
       import type { Session, User } from '@supabase/supabase-js'

       export interface AuthSession {
         session: Session
         user: User
         pool: 'internal' | 'external'
         roles: string[]
         tenantId: string | null
       }

       export interface AuthGuardOptions {
         supabaseUrl: string
         supabaseAnonKey: string
         loginPath?: string
         requiredPool?: 'internal' | 'external'
       }
       ```

    2. Create `packages/auth/src/guard.ts`:
       - Import `redirect` from `@tanstack/react-router`
       - Import `createSupabaseServerClient` from `./server`
       - Import `getRequest` from `@tanstack/react-start/server`
       - Import `AuthGuardOptions`, `AuthSession` from `./types`
       - Export `async function authGuard(opts: AuthGuardOptions): Promise<AuthSession>`
       - Implementation:
         a. Call `getRequest()` to get the current Request
         b. Call `createSupabaseServerClient({ request, supabaseUrl: opts.supabaseUrl, supabaseAnonKey: opts.supabaseAnonKey })`
         c. Call `client.auth.getSession()` to check for valid session
         d. If no session: `throw redirect({ to: opts.loginPath ?? '/login' })`
         e. Extract from session.user.app_metadata: pool, roles, tenant_id (set by custom access token hook from Phase 2)
         f. If `opts.requiredPool` is set and doesn't match: `throw redirect({ to: opts.loginPath ?? '/login' })`
         g. Return `{ session, user: session.user, pool, roles, tenantId }`
       - CRITICAL: This guard calls createSupabaseServerClient which creates a client with the user's JWT from cookies. All subsequent Supabase queries through this client will enforce RLS policies.

    3. Create `packages/auth/src/session.ts`:
       - Export `async function getServerSession(opts: { supabaseUrl: string; supabaseAnonKey: string }): Promise<AuthSession | null>`
         - Same as authGuard but returns null instead of redirecting
         - For use in loaders where you want optional auth (e.g., website pages)
       - Export `function hasPermission(session: AuthSession, permission: string): boolean`
         - Checks session.roles against role_permissions (client-side check)
         - Note: real authorization is enforced by RLS, this is for UI gating only

    4. Update `packages/auth/src/index.ts`:
       ```typescript
       export { createSupabaseServerClient } from './server'
       export { createSupabaseBrowserClient } from './client'
       export { authGuard } from './guard'
       export { getServerSession, hasPermission } from './session'
       export type { AuthSession, AuthGuardOptions } from './types'
       ```

    5. Update `packages/auth/package.json` -- add peerDependencies:
       ```json
       "peerDependencies": {
         "@tanstack/react-router": "^1.168.0",
         "@tanstack/react-start": "^1.167.12"
       }
       ```

       ```json

       "./session": "./src/session.ts",
       "./types": "./src/types.ts"
       ```
