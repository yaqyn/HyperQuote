    **3. Create `apps/portal/src/routes/_portal.tsx`** (per PORT-01):
    This is the auth layout route. ALL portal child routes nest under this.

    - Import: Outlet, createFileRoute from `@tanstack/react-router`, redirect from `@tanstack/react-router`
    - Import: getServerSession from `@hyperquote/auth` (NOT authGuard -- we need external redirect)

    **Critical: Do NOT use authGuard() here.** The shared authGuard uses `redirect({ to: ... })` which only works for same-app routes. Portal needs cross-app redirect to the website. Instead:

    ```typescript
    export const Route = createFileRoute('/_portal')({
      beforeLoad: async ({ location }) => {
        // Server-side auth check via getServerSession
        const session = await getServerSession({
          supabaseUrl: SUPABASE_URL,
          supabaseAnonKey: SUPABASE_ANON_KEY,
        })

        if (!session) {
          // Cross-app redirect to website login
          const redirectUrl = `${WEBSITE_URL}?login=portal&redirect=${encodeURIComponent(location.href)}`
          throw redirect({ href: redirectUrl })
        }

        // Check pool -- portal requires external pool
        const pool = session.user.app_metadata?.pool ?? 'external'
        if (pool !== 'external') {
          // Internal-only user -- return flag for error page rendering
          return { auth: null, isInternalUser: true }
        }

        const roles = (session.user.app_metadata?.roles as string[]) ?? []
        const tenantId = (session.user.app_metadata?.tenant_id as string) ?? null

        return {
          auth: {
            session,
            user: session.user,
            pool: pool as 'internal' | 'external',
            roles,
            tenantId,
          },
          isInternalUser: false,
        }
      },
      component: PortalLayout,
    })
    ```

    **NOTE on getServerSession:** Check if `getServerSession` in `@hyperquote/auth` accepts `{ supabaseUrl, supabaseAnonKey }` or needs different args. Read the actual function signature from `packages/auth/src/session.ts`. If it has a different API, adapt accordingly. The key is: get the session, check if it exists, check the pool. If `getServerSession` doesn't support this pattern, use `createSupabaseServerClient` directly:
    ```typescript
    import { createSupabaseServerClient } from '@hyperquote/auth'
    import { getRequest } from '@tanstack/react-start/server'

    // In beforeLoad:
    const request = getRequest()
    const { client } = createSupabaseServerClient({
      request,
      supabaseUrl: SUPABASE_URL,
      supabaseAnonKey: SUPABASE_ANON_KEY,
    })
    const { data: { session } } = await client.auth.getSession()
    ```
