    CREATE FUNCTION update_product_search_vector() RETURNS TRIGGER AS $$
    BEGIN
      NEW.search_vector := to_tsvector('english', coalesce(NEW.name, '') || ' ' || coalesce(NEW.name_ar, '') || ' ' || coalesce(NEW.brand, '') || ' ' || coalesce(NEW.category::text, '') || ' ' || coalesce(NEW.sku, ''));

    END;
    $$ LANGUAGE plpgsql;

      BEFORE INSERT OR UPDATE ON products
      FOR EACH ROW EXECUTE FUNCTION update_product_search_vector();
