   export type ActivityEntryType = 'comment_internal' | 'comment_external' | 'system' | 'mention' | 'handoff'

   export interface ActivityEntry {
     id: string
     entityType: string  // 'order' | 'quote' | 'customer' | 'delivery' | 'ticket' etc.
     entityId: string
     type: ActivityEntryType
     actorId: string
     actorName: string
     actorRole?: string
     body: string        // plain text, may contain @mentions as @[name](userId)
     createdAt: string   // ISO datetime
     metadata?: Record<string, unknown>  // type-specific data (e.g., handoff target, system event details)
   }

   export interface MentionTarget {
     id: string
     name: string
     role: string
     department?: string
   }

   export interface HandoffStatus {

     entityId: string
     assignedTo: string
     assignedAt: string        // ISO datetime
     acknowledgedAt: string | null
     escalatedAt: string | null
     escalationTarget: string | null  // department manager

1. Create `apps/internal/src/hooks/useActivityFeed.ts`:
   - Props: `{ entityType: string; entityId: string }`
   - Uses `useQuery({ queryKey: ['activity', entityType, entityId], queryFn: () => getActivityEntries({ data: { entityType, entityId } }), staleTime: 30_000 })`
   - Uses `useMutation` for `postComment` with `onSuccess` invalidating `['activity', entityType, entityId]`
   - Returns `{ entries, isLoading, postComment: mutate, isPosting }`

2. Create `apps/internal/src/components/activity-feed/ActivityItem.tsx`:
   - Props: `{ entry: ActivityEntry }`
   - Renders based on `entry.type`:
     - `comment_internal`: Blue left border (`border-s-2 border-[var(--color-primary)]`), actor name + role, body text, timestamp
     - `comment_external`: Gray left border (`border-s-2 border-[var(--color-border)]`), actor name, body text, timestamp
     - `system`: No border, italic text, system icon (Activity from lucide), muted styling
     - `mention`: Similar to comment_internal but with @mention badges highlighted
     - `handoff`: Special styling with Hot Potato icon (Flame from lucide), assignment info
   - @mention rendering: parse `@[name](userId)` in body text, replace with inline badge:
     `<span className="inline-flex items-center gap-0.5 px-1 rounded bg-[var(--color-primary)]/10 text-[var(--color-primary)] text-xs font-medium">@{name}</span>`
   - Timestamp: Geist Mono font, `text-xs text-[var(--color-text-muted)]`, relative time (e.g., "2m ago", "1h ago", "Yesterday")
   - Actor avatar: circle with initials (first letter of name), 28px, `bg-[var(--color-primary)]/10 text-[var(--color-primary)] text-xs font-semibold`
   - Layout: avatar start-aligned, content flex-col with name+role on first line, body below, timestamp end-aligned

3. Create `apps/internal/src/components/activity-feed/MentionInput.tsx`:
   - Props: `{ onSubmit: (body: string, isInternal: boolean) => void; isSubmitting?: boolean }`
   - Text input area: `<textarea>` (NOT React Aria TextField -- need multi-line with @mention detection)
   - @mention detection: when user types `@`, show a popover (CSS transition, NOT Motion) with search results from `searchMentionTargets`
   - Popover: positioned above cursor, `backdrop-blur-xl bg-white/90 dark:bg-black/90 shadow-md rounded-lg`
   - Each mention result: `<Button>` with name + role, on press inserts `@[name](userId)` into textarea at cursor position
   - Debounced search: 200ms after `@` trigger, min 1 char after `@`
   - Internal/External toggle: Two radio-style buttons above the textarea
     - "Internal" (blue highlight when active): `bg-[var(--color-primary)] text-white`
     - "External" (gray when active): `bg-[var(--color-border)] text-[var(--color-text)]`
     - Default: Internal
   - Submit button: `<Button>` from react-aria-components, text "Post" (localized), `bg-[var(--color-primary)] text-white rounded-lg px-4 py-1.5 text-sm`
   - On focus: dispatch `focusInput` to keyboard scope store
   - On blur (when not selecting mention): dispatch `blurInput` to keyboard scope store
   - Submit on Ctrl+Enter (in addition to button click)

4. Create `apps/internal/src/components/activity-feed/ActivityFeed.tsx`:

   - Auto-scroll to bottom on new entries (useEffect with ref to scroll container)

     - "Acknowledge" button if current user is the assignee
   - Loading state: 3 skeleton bars (height 40px, rounded, shimmer animation)
   - Empty state: "No activity yet" with subtle icon
   - Header: "Activity" title (text-sm font-semibold text-[var(--color-text-muted)] uppercase tracking-wide)
